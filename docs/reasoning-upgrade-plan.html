Key Insight: Parallel Reasoning = Free Depth

Because all 6 reasoning calls fire in parallel (Promise.all), the wall-clock time is the slowest single call (~3s), not the sum. You get 6x more analysis for the cost of 1 sequential call. The fast explanation (qwen-plus, 1.1s) runs after reasoning completes and can reference their findings.

ReasoningEngine Class (agents/reasoner.ts)

import { LLMRouter } from '../llm/router.js';
import { getModel } from '../llm/model-assignments.js';

export interface ReasoningAnalysis {
  thesis: ThesisResult;
  risk: RiskResult;
  scenarios: ScenarioResult;
  exit: ExitResult;
  regime: RegimeResult;
  sizing: SizingResult;
  meta: {
    totalConviction: number;      // 1-10, weighted
    overallGrade: string;         // A/B/C/D
    shouldPublish: boolean;       // conviction > 6 AND regime > 0.5
    totalCostUsd: number;
    wallClockMs: number;
  };
}

export class ReasoningEngine {
  private router: LLMRouter;

  constructor(router: LLMRouter) {
    this.router = router;
  }

  async analyze(data: ReasoningInput): Promise<ReasoningAnalysis> {
    const t0 = Date.now();
    const model = getModel('reasoning-thesis'); // qwen3-max

    // All 6 reasoning calls in parallel
    const [thesis, risk, scenarios, exit, regime, sizing] = await Promise.all([
      this.reasonThesis(model, data),
      this.reasonRisk(model, data),
      this.reasonScenarios(model, data),
      this.reasonExit(model, data),
      this.reasonRegime(model, data),
      this.reasonSizing(model, data),
    ]);

    // Compute meta scores
    const totalConviction = this.computeConviction(thesis, regime);
    const overallGrade = this.computeGrade(totalConviction, risk);

    return {
      thesis, risk, scenarios, exit, regime, sizing,
      meta: {
        totalConviction,
        overallGrade,
        shouldPublish: totalConviction > 6 && regime.alignmentScore > 0.5,
        totalCostUsd: this.router.getUsage().totalCost,
        wallClockMs: Date.now() - t0,
      }
    };
  }
}

16. Scenario Detection — How Engine Routes to Right Analysis

Automatic Scenario Classification (Deterministic, pre-LLM)

function classifyScenario(data: EngineInput): Scenario[] {
  const scenarios: Scenario[] = [];

  // High IV premium selling
  if (data.ivRank > 50 || data.atmIv > 35 || data.ivRvRatio > 1.2)
    scenarios.push('HIGH_IV_SELL');

  // Low IV vol buying
  if (data.ivRank < 20 || data.atmIv < 20 || data.ivRvRatio < 0.8)
    scenarios.push('LOW_IV_BUY');

  // Strong trend
  if (data.adx > 30 && data.trendScore !== 0.5)
    scenarios.push('STRONG_TREND');

  // Range-bound
  if (data.adx < 20 && data.bandWidth >= 3 && data.bandWidth <= 15)
    scenarios.push('RANGE_BOUND');

  // Pre-earnings
  if (data.earningsDaysOut > 0 && data.earningsDaysOut <= data.dte)
    scenarios.push('PRE_EARNINGS');

  // Post-earnings
  if (data.daysSinceEarnings <= 5)
    scenarios.push('POST_EARNINGS');

  // Gamma breakout
  if (data.spot > data.callWall || data.spot < data.putWall)
    scenarios.push('GAMMA_BREAKOUT');

  // Macro shock
  if (data.vix > 25 || data.spyChange < -0.02)
    scenarios.push('MACRO_SHOCK');

  // Mean reversion
  if (data.rsi < 30 || data.rsi > 70)
    scenarios.push('MEAN_REVERSION');

  // Wide band (BWB territory)
  if (data.bandWidth > 15 && data.bandWidth <= 40)
    scenarios.push('WIDE_BAND_BWB');

  // Dividend within DTE
  if (data.exDivDays > 0 && data.exDivDays <= data.dte)
    scenarios.push('DIVIDEND_RISK');

  // Low liquidity
  if (data.bidAskSpreadPct > 0.10 || data.strikeOI < 100)
    scenarios.push('LOW_LIQUIDITY');

  // Term structure opportunity
  if (data.frontIv / data.backIv > 1.3)
    scenarios.push('TERM_STRUCTURE');

  return scenarios;  // Can be MULTIPLE scenarios simultaneously
}

// Route to appropriate analysis tasks based on detected scenarios
function getRequiredAnalysis(scenarios: Scenario[]): AnalysisTask[] {
  const tasks: Set<AnalysisTask> = new Set();

  for (const scenario of scenarios) {
    SCENARIO_ANALYSIS_MAP[scenario].forEach(task => tasks.add(task));
  }

  // De-duplicate and prioritize (max 8 reasoning calls to control cost)
  return prioritize([...tasks], MAX_REASONING_CALLS);
}
Key Design Decision: Dynamic Task Selection

Instead of always running all 6 reasoning tasks, the system detects which scenarios apply and selects only the RELEVANT reasoning prompts. This means:

  • Range-bound AAPL: 4 reasoning calls (range persistence, breakout direction, wall strength, wing width) = ~$0.012
  • Pre-earnings AMZN: 7 reasoning calls (full earnings analysis suite) = ~$0.021
  • Simple high-IV sell: 4 reasoning calls (IV persistence, strike width, vega risk, catalyst) = ~$0.012
  • Macro shock + existing position: 6+5 = 11 reasoning calls (capped at 8 via priority) = ~$0.024

Average across all scenarios: ~5 reasoning calls per request, all parallel. Cost: ~$0.015. Latency: 2.5-3s.
